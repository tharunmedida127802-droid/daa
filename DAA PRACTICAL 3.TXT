#include <iostream>
#include <vector>
#include <chrono>
#include <random>
#include <iomanip>

using namespace std;
using namespace chrono;


// --------------------------------------------------
// COMMON HEAPIFY FUNCTION
// maxMode = true  -> Max Heap
// maxMode = false -> Min Heap
// --------------------------------------------------

void heapify(vector<int>& a, int size, int position, bool maxMode)
{
    while (true)
    {
        int selected = position;
        int left = 2 * position + 1;
        int right = 2 * position + 2;

        if (maxMode)
        {
            if (left < size && a[left] > a[selected])
                selected = left;

            if (right < size && a[right] > a[selected])
                selected = right;
        }
        else
        {
            if (left < size && a[left] < a[selected])
                selected = left;

            if (right < size && a[right] < a[selected])
                selected = right;
        }

        if (selected == position)
            break;

        swap(a[position], a[selected]);
        position = selected;
    }
}


// --------------------------------------------------
// BUILD HEAP
// --------------------------------------------------

void makeHeap(vector<int>& a, bool maxMode)
{
    int size = static_cast<int>(a.size());

    for (int i = size / 2 - 1; i >= 0; i--)
    {
        heapify(a, size, i, maxMode);
    }
}


// --------------------------------------------------
// HEAP SORT
// --------------------------------------------------

void heapSort(vector<int>& a, bool maxMode)
{
    int size = static_cast<int>(a.size());

    makeHeap(a, maxMode);

    for (int end = size - 1; end >= 1; end--)
    {
        swap(a[0], a[end]);
        heapify(a, end, 0, maxMode);
    }

    // Min Heap gives descending order,
    // so reverse it to get ascending order.
    if (!maxMode)
    {
        int start = 0;
        int finish = size - 1;

        while (start < finish)
        {
            swap(a[start], a[finish]);
            start++;
            finish--;
        }
    }
}


// --------------------------------------------------
// CHECK SORTING
// --------------------------------------------------

bool checkSorted(const vector<int>& a)
{
    for (int i = 1; i < static_cast<int>(a.size()); i++)
    {
        if (a[i - 1] > a[i])
            return false;
    }

    return true;
}


// --------------------------------------------------
// DISPLAY FIRST FEW ELEMENTS
// --------------------------------------------------

void displaySample(const vector<int>& a)
{
    int limit = min(10, static_cast<int>(a.size()));

    for (int i = 0; i < limit; i++)
    {
        cout << a[i] << " ";
    }

    if (a.size() > 10)
        cout << "...";

    cout << endl;
}


// --------------------------------------------------
// MAIN FUNCTION
// --------------------------------------------------

int main()
{
    int n;

    cout << "Enter the number of elements: ";
    cin >> n;

    if (n <= 0)
    {
        cout << "Please enter a positive number." << endl;
        return 0;
    }

    // Generate random elements
    vector<int> original(n);

    random_device seed;
    mt19937 engine(seed());
    uniform_int_distribution<int> randomNumber(100, 99999);

    for (int i = 0; i < n; i++)
    {
        original[i] = randomNumber(engine);
    }

    // Create two copies of the same data
    vector<int> maxVersion = original;
    vector<int> minVersion = original;


    // ==================================================
    // MAX HEAP SORT
    // ==================================================

    auto maxBegin = steady_clock::now();

    heapSort(maxVersion, true);

    auto maxEnd = steady_clock::now();

    auto maxDuration = maxEnd - maxBegin;


    // ==================================================
    // MIN HEAP SORT
    // ==================================================

    auto minBegin = steady_clock::now();

    heapSort(minVersion, false);

    auto minEnd = steady_clock::now();

    auto minDuration = minEnd - minBegin;


    // ==================================================
    // DISPLAY RESULTS
    // ==================================================

    cout << "\n========================================";
    cout << "\n           HEAP SORT RESULTS";
    cout << "\n========================================";


    cout << "\n\nMAX HEAP SORT";
    cout << "\n----------------------------------------";

    cout << "\nNanoseconds  : "
         << duration_cast<nanoseconds>(maxDuration).count()
         << " ns";

    cout << "\nMicroseconds : "
         << duration_cast<microseconds>(maxDuration).count()
         << " us";

    cout << "\nMilliseconds : "
         << duration_cast<milliseconds>(maxDuration).count()
         << " ms";

    cout << "\nSeconds      : "
         << fixed << setprecision(9)
         << duration<double>(maxDuration).count()
         << " sec";

    cout << "\nSorted correctly : "
         << (checkSorted(maxVersion) ? "YES" : "NO");

    cout << "\nFirst values : ";
    displaySample(maxVersion);


    cout << "\n\nMIN HEAP SORT";
    cout << "\n----------------------------------------";

    cout << "\nNanoseconds  : "
         << duration_cast<nanoseconds>(minDuration).count()
         << " ns";

    cout << "\nMicroseconds : "
         << duration_cast<microseconds>(minDuration).count()
         << " us";

    cout << "\nMilliseconds : "
         << duration_cast<milliseconds>(minDuration).count()
         << " ms";

    cout << "\nSeconds      : "
         << fixed << setprecision(9)
         << duration<double>(minDuration).count()
         << " sec";

    cout << "\nSorted correctly : "
         << (checkSorted(minVersion) ? "YES" : "NO");

    cout << "\nFirst values : ";
    displaySample(minVersion);

    cout << "\n\n========================================\n";

    return 0;
}


/*
=========================================================
                     ALGORITHM
               MAX HEAP AND MIN HEAP SORT
=========================================================

1. Start.

2. Read the number of elements N from the user.

3. Create an array of N elements.

4. Generate random values and store them in the array.

5. Create two copies of the original array:
      - One for Max Heap Sort.
      - One for Min Heap Sort.

---------------------------------------------------------
             MAX HEAP SORT ALGORITHM
---------------------------------------------------------

6. Build a Max Heap from the first array.

7. Start from the last element of the heap.

8. Swap the root element with the last element.

9. Reduce the heap size by one.

10. Apply Max Heapify to restore the Max Heap property.

11. Repeat steps 8 to 10 until only one element remains.

12. The array is now sorted in ascending order.

---------------------------------------------------------
             MIN HEAP SORT ALGORITHM
---------------------------------------------------------

13. Build a Min Heap from the second array.

14. Start from the last element of the heap.

15. Swap the root element with the last element.

16. Reduce the heap size by one.

17. Apply Min Heapify to restore the Min Heap property.

18. Repeat steps 15 to 17 until only one element remains.

19. The result is in descending order.

20. Reverse the array using element swapping to obtain
    ascending order.

---------------------------------------------------------
                 TIME COMPARISON
---------------------------------------------------------

21. Record the starting time before Max Heap Sort.

22. Record the ending time after Max Heap Sort.

23. Calculate the execution time.

24. Repeat the same process for Min Heap Sort.

25. Display execution time in:
      - Nanoseconds
      - Microseconds
      - Milliseconds
      - Seconds

26. Check whether both arrays are correctly sorted.

27. Display the results.

28. Stop.


Time Complexity:
    Best Case    : O(N log N)
    Average Case : O(N log N)
    Worst Case   : O(N log N)

Space Complexity:
    O(1) auxiliary space for Heap Sort.

=========================================================
*/
