#include <iostream>
#include <vector>
#include <stack>
#include <queue>
#include <chrono>

using namespace std;

class GraphTraversal
{
private:
    int vertices;
    vector<vector<int>> matrix;

public:
    GraphTraversal(int n)
    {
        vertices = n;
        matrix.resize(n, vector<int>(n, 0));
    }

    void connect(int a, int b)
    {
        if (a < 0 || a >= vertices || b < 0 || b >= vertices)
        {
            cout << "Invalid edge! Vertex must be between 0 and "
                 << vertices - 1 << endl;
            return;
        }

        matrix[a][b] = 1;
        matrix[b][a] = 1;
    }

    void showGraph()
    {
        cout << "\nAdjacency Matrix:\n";

        for (int i = 0; i < vertices; i++)
        {
            for (int j = 0; j < vertices; j++)
            {
                cout << matrix[i][j] << " ";
            }
            cout << endl;
        }
    }

    void depthFirst(int start)
    {
        if (start < 0 || start >= vertices)
        {
            cout << "Invalid starting vertex!";
            return;
        }

        vector<bool> visited(vertices, false);
        stack<int> s;

        s.push(start);

        while (!s.empty())
        {
            int current = s.top();
            s.pop();

            if (visited[current])
                continue;

            visited[current] = true;

            cout << current << " ";

            for (int i = vertices - 1; i >= 0; i--)
            {
                if (matrix[current][i] == 1 && !visited[i])
                {
                    s.push(i);
                }
            }
        }
    }

    void breadthFirst(int start)
    {
        if (start < 0 || start >= vertices)
        {
            cout << "Invalid starting vertex!";
            return;
        }

        vector<bool> visited(vertices, false);
        queue<int> q;

        visited[start] = true;
        q.push(start);

        while (!q.empty())
        {
            int current = q.front();
            q.pop();

            cout << current << " ";

            for (int i = 0; i < vertices; i++)
            {
                if (matrix[current][i] == 1 && !visited[i])
                {
                    visited[i] = true;
                    q.push(i);
                }
            }
        }
    }
};

int main()
{
    int n, edges;

    cout << "Enter number of vertices: ";
    cin >> n;

    if (n <= 0)
    {
        cout << "Number of vertices must be greater than 0.";
        return 0;
    }

    GraphTraversal graph(n);

    cout << "Enter number of edges: ";
    cin >> edges;

    cout << "Enter edges (u v):\n";

    for (int i = 0; i < edges; i++)
    {
        int u, v;
        cin >> u >> v;

        graph.connect(u, v);
    }

    int start;

    cout << "Enter starting vertex: ";
    cin >> start;

    if (start < 0 || start >= n)
    {
        cout << "Invalid starting vertex!";
        return 0;
    }

    graph.showGraph();

    cout << "\nDFS Traversal: ";

    auto dfsStart = chrono::high_resolution_clock::now();

    graph.depthFirst(start);

    auto dfsEnd = chrono::high_resolution_clock::now();

    auto dfsTime =
        chrono::duration_cast<chrono::nanoseconds>
        (dfsEnd - dfsStart);

    cout << "\nDFS Time: " << dfsTime.count() << " ns";


    cout << "\n\nBFS Traversal: ";

    auto bfsStart = chrono::high_resolution_clock::now();

    graph.breadthFirst(start);

    auto bfsEnd = chrono::high_resolution_clock::now();

    auto bfsTime =
        chrono::duration_cast<chrono::nanoseconds>
        (bfsEnd - bfsStart);

    cout << "\nBFS Time: " << bfsTime.count() << " ns";

    return 0;
}

/*
====================================================
                DFS ALGORITHM
====================================================

1. Start from the given vertex.
2. Create a visited list.
3. Mark all vertices as not visited.
4. Put the starting vertex into a stack.
5. Take the top vertex from the stack.
6. Mark it as visited and print it.
7. Add its unvisited connected vertices to the stack.
8. Repeat until the stack becomes empty.
9. Stop.


====================================================
                BFS ALGORITHM
====================================================

1. Start from the given vertex.
2. Create a visited list.
3. Mark all vertices as not visited.
4. Put the starting vertex into a queue.
5. Mark the starting vertex as visited.
6. Take the first vertex from the queue.
7. Print the vertex.
8. Add its unvisited connected vertices to the queue.
9. Repeat until the queue becomes empty.
10. Stop.
====================================================
*/
